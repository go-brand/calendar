---
title: React Calendar
description: Month, week, and day calendars for React, built on Temporal.
---

You bring the markup and styles. Calendar handles the date, the view, navigation, the timezone, and where each event goes.

```tsx
import type { Temporal } from '@js-temporal/polyfill'
import { useCreateCalendar } from '@gobrand/react-calendar'

type Booking = {
  id: string
  guest: string
  date: Temporal.PlainDate
}

function BookingCalendar({ bookings }: { bookings: Booking[] }) {
  const calendar = useCreateCalendar({
    views: { month: true },
    items: bookings,
    getDate: booking => booking.date,
  })

  return calendar.activeView.weeks.map(week => (
    <div key={week[0].id}>
      {week.map(day => (
        <div key={day.id}>
          {day.date.day}: {day.items.map(booking => booking.guest).join(', ')}
        </div>
      ))}
    </div>
  ))
}
```

Only `month` is enabled, so `activeView` is a month. Add `week` or `day` when you need them.
