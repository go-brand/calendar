---
title: useCalendarView
description: Put events into one view of a shared calendar.
---

`useCalendarView` reads the calendar from `CalendarProvider`, or from a `calendar` you pass. TypeScript infers the event type from `items`.

```tsx
import type { Temporal } from '@js-temporal/polyfill'
import { useCalendarView } from '@gobrand/react-calendar'

type Meeting = {
  id: string
  title: string
  startsAt: Temporal.ZonedDateTime
}

function WeekContent({ meetings }: { meetings: Meeting[] }) {
  const week = useCalendarView({
    view: 'week',
    items: meetings,
    getDate: meeting => meeting.startsAt.toPlainDate(),
    getStart: meeting => meeting.startsAt,
  })

  return week.days.map(day => (
    <div key={day.id}>
      {day.date.toString()}: {day.items.map(meeting => meeting.title).join(', ')}
    </div>
  ))
}
```

`getDate` is required with `items`. Add `getStart` to place events in time slots. The view you ask for is the view you get: `view: 'week'` stays a week even when the calendar switches to month.
